Usage Example

How to use the platform frames API

// Basic usage const frameHTML = buildContextFrame('twitter', { title: 'Amazing Article', description: 'Check out this great content', image: 'https://example.com/image.jpg', domain: 'example.com', author: 'Developer', timeAgo: '2h' }, 'dark'); // Enhanced rendering with validation and caching const result = PlatformFramesRenderer.buildEnhancedFrame( 'linkedin', { title: 'Professional Post', description: 'Industry insights and trends', image: 'https://example.com/image.jpg', domain: 'linkedin.com' }, 'light', { useCache: true, validate: true, accessibility: true, responsive: true } ); // Render directly into container PlatformFramesRenderer.renderFrame( document.getElementById('container'), 'slack', { title: 'Check this out!', domain: 'example.com' }, 'dark' ); // Theme management if (hasThemeSupport('twitter')) { const vars = getThemeVars('twitter', 'dark'); console.log('Theme variables:', vars); }
